import { readFileSync, writeFileSync } from 'node:fs';
import { join, resolve, dirname } from 'node:path';
import { randomUUID } from 'node:crypto';
import { spawnSync } from 'node:child_process';
import { fileURLToPath } from 'node:url';
import { snapshot, captureResult } from '../check.mjs';

const trialDir = dirname(fileURLToPath(import.meta.url));
const TARGET = 'CHECKOUT_ROOT';
if (resolve(trialDir, '../../..') !== TARGET) throw new Error('driver is outside the target checkout');
const PROCESS = 'i542_migrate_spec_id_refs';
const ID = 'SPEC_single_producer_constraint';
const HEADING = '### 15.1 単一生成元制約';
const REPL = `[[${ID}]]`;
const REF = /§15\.1(?![0-9.])/g;
const ALLOWED = ['docs/spec/spec.md', '.pfdsl/roadmap.pfdsl'];
const RUN_ID = randomUUID();
const commands = [];
const assertions = [];

function run(args) {
  const r = spawnSync(process.execPath, args, { cwd: TARGET, encoding: 'utf8' });
  const rec = { args, status: r.status, stdout: r.stdout ?? '', stderr: r.stderr ?? '' };
  commands.push(rec);
  return rec;
}
function assert(name, ok) {
  assertions.push({ name, ok: Boolean(ok) });
  if (!ok) throw new Error(`assertion failed: ${name}`);
}
function saveChecks(status) {
  const body = { schemaVersion: 0, runId: RUN_ID, processId: PROCESS, status, assertions, commands };
  writeFileSync(join(trialDir, 'checks.json'), JSON.stringify(body, null, 2) + '\n');
}
const readyIds = v => (Array.isArray(v) ? v : v.ready ?? []).map(e => e.id);
const sameVersions = (a, b) => {
  const ks = Object.keys(b);
  return ks.length === 3 && Object.keys(a).length === 3 && ks.every(k => a[k] === b[k]);
};
function sectionOf(lines, start) {
  const end = lines.findIndex((l, i) => i > start && /^#{1,3} /.test(l));
  return { body: lines.slice(start + 1, end < 0 ? lines.length : end), next: end < 0 ? null : lines[end] };
}

try {
  const preSnapshot = JSON.parse(readFileSync(join(trialDir, 'snapshot.json'), 'utf8'));
  const frozenReady = JSON.parse(readFileSync(join(trialDir, 'ready.json'), 'utf8'));
  assert('frozen ready set contains process', readyIds(frozenReady).includes(PROCESS));
  const cli = run(['packages/cli/dist/cli.js', 'status', 'ready', '.pfdsl/roadmap.pfdsl', '--json']);
  assert('cli status ready exits 0', cli.status === 0);
  const liveReady = JSON.parse(cli.stdout);
  assert('cli ready ok=true', liveReady.ok === true);
  assert('cli ready set contains process', readyIds(liveReady).includes(PROCESS));
  assert('captured versions match frozen versions', sameVersions(captureResult(preSnapshot, TARGET).versions, preSnapshot.versions));

  const specPath = join(TARGET, 'docs/spec/spec.md');
  const preText = readFileSync(specPath, 'utf8');
  const preLines = preText.split('\n');
  const absent = run(['scripts/get-spec-id.mjs', ID]);
  assert('new id not defined', absent.status === 1 && /not defined|undefined|未定義/i.test(absent.stdout + absent.stderr));
  const mint = run(['scripts/mint-check.mjs', 'single_producer_constraint']);
  assert('mint-check exits 0', mint.status === 0);
  const hIdx = preLines.flatMap((l, i) => (l.replace(/\r$/, '') === HEADING ? [i] : []));
  assert('heading occurs once', hIdx.length === 1);
  assert('two section references', (preText.match(REF) ?? []).length === 2);
  assert('new id absent from spec', !preText.includes(ID));

  const meta = run(['packages/cli/dist/cli.js', 'meta', 'set', '.pfdsl/roadmap.pfdsl', 'spec_id_refs', 'status', 'wip']);
  assert('meta set status wip exits 0', meta.status === 0);

  const lines = [...preLines];
  lines[hIdx[0]] = lines[hIdx[0]].replace(HEADING, `${HEADING} (${ID})`);
  const postText = lines.join('\n').replace(REF, REPL);
  writeFileSync(specPath, postText);

  const chk = run(['scripts/check-spec-ids.mjs']);
  assert('check-spec-ids exits 0', chk.status === 0);
  const get = run(['scripts/get-spec-id.mjs', ID]);
  assert('get-spec-id exits 0', get.status === 0);
  const postLines = postText.split('\n');
  const pIdx = postLines.findIndex(l => l.startsWith(HEADING));
  const preSec = sectionOf(preLines, hIdx[0]);
  const postSec = sectionOf(postLines, pIdx);
  assert('section body unchanged except references', postSec.body.join('\n') === preSec.body.join('\n').replace(REF, REPL));
  assert('section boundary unchanged', postSec.next === preSec.next);
  assert('fetched block contains section body', postSec.body.filter(l => l.trim()).every(l => get.stdout.includes(l.trim())));
  assert('fetched block stops before next section', postSec.next === null || !get.stdout.includes(postSec.next.trim()));
  assert('no old references remain', (postText.match(REF) ?? []).length === 0);
  assert('replacement sites exactly two', postText.split(REPL).length - 1 === 2);

  const post = captureResult(preSnapshot, TARGET);
  const pm = preSnapshot.manifest ?? {};
  const qm = post.manifest ?? {};
  const changedPaths = [...new Set([...Object.keys(pm), ...Object.keys(qm)])]
    .filter(k => JSON.stringify(pm[k]) !== JSON.stringify(qm[k])).sort();
  assert('changed paths within allowed targets', changedPaths.length > 0 && changedPaths.every(p => ALLOWED.includes(p)));
  assert('spec.md changed', changedPaths.includes('docs/spec/spec.md'));
  const postPath = join(trialDir, 'trial-snapshot.json');
  writeFileSync(postPath, JSON.stringify(post, null, 2) + '\n');
  saveChecks('RECORDED');

  const evidencePath = join(trialDir, 'evidence.json');
  writeFileSync(evidencePath, JSON.stringify({
    schemaVersion: 0,
    runId: RUN_ID,
    processId: PROCESS,
    workItemIds: ['542'],
    versions: preSnapshot.versions,
    resultVersions: post.versions,
    changedPaths,
    transitions: [{ artifactId: 'spec_id_refs', from: 'todo', to: 'wip' }],
    acceptanceEvidence: ['experiments/evidence-roundtrip/safe-mode-20261026/checks.json'],
    deviations: [{ type: 'unchanged', reason: 'No model revision is requested: issue #542 explicitly allows staged migration; only the section 15.1 cluster was migrated and the complete artifact criteria remain unmet, so spec_id_refs stays wip.' }],
  }, null, 2) + '\n');

  const verdict = run([join(TARGET, 'experiments/evidence-roundtrip/check.mjs'), evidencePath, postPath, TARGET]);
  assert('checker exits 0', verdict.status === 0);
  writeFileSync(join(trialDir, 'result.json'), verdict.stdout);
  assert('checker status PASS', JSON.parse(verdict.stdout).status === 'PASS');
  saveChecks('PASS');
  console.log(`PASS ${PROCESS} spec_id_refs wip (run ${RUN_ID})`);
} catch (err) {
  saveChecks('FAIL');
  console.error(err.message);
  process.exitCode = 1;
}
