import assert from 'node:assert/strict';
import { createRequire } from 'node:module';
import { writeFileSync } from 'node:fs';
import { join } from 'node:path';
import vm from 'node:vm';
const repo = '/path/to/pfdsl';
const out = '/tmp/pfdsl-audit/render-editor';
const req = createRequire(join(repo, 'packages/vscode-extension/package.json'));
const { build } = req('esbuild');
const { JSDOM } = req('jsdom');
const core = await import(join(repo,'packages/core/dist/index.js'));
const graphviz = await import(join(repo,'packages/graphviz-exporter/dist/index.js'));
const previewEngine = await import(join(repo,'packages/preview-engine/dist/index.js'));
async function load(file, mocks = {}, globals = {}) {
  const bundle = await build({ entryPoints: [join(repo, file)], bundle: true, packages: 'external', platform: 'node', format: 'cjs', write: false, logLevel: 'silent' });
  const module = { exports: {} };
  vm.runInNewContext(bundle.outputFiles[0].text, {
    module, exports: module.exports, console, Buffer, process,
    require: name => name === '@pfdsl/core' ? core : name === '@pfdsl/graphviz-exporter' ? graphviz : name === '@pfdsl/preview-engine' ? previewEngine : Object.hasOwn(mocks, name) ? mocks[name] : req(name), ...globals,
  }, { filename: file });
  return module.exports;
}
function uri(path) { return { scheme: 'file', path, fsPath: path, toString: () => `file://${path}` }; }
function document(path, source) { return { uri: uri(path), version: 1, languageId: 'pfdsl', getText: () => source }; }
function vscodeHarness(doc) {
  const commands = new Map(), panels = [], opened = [], messages = [];
  const vscode = {
    Uri: { file: uri, joinPath: (base, ...parts) => uri(join(base.fsPath, ...parts)), parse: value => ({ toString: () => value }) },
    ViewColumn: { Beside: 2 }, ExtensionMode: { Development: 2 }, FileType: { Directory: 2 },
    commands: { registerCommand: (id, fn) => { commands.set(id, fn); return {}; }, executeCommand: async id => commands.get(id)?.() },
    window: {
      activeTextEditor: { document: doc, selection: { active: { line: 0, character: 0 } } }, visibleTextEditors: [],
      createWebviewPanel: () => {
        const panel = { webview: { cspSource: 'mock:', asWebviewUri: u => u, postMessage: msg => messages.push(msg), onDidReceiveMessage: fn => { panel.receive = fn; return {}; } }, onDidChangeViewState: fn => { panel.changeViewState = fn; return {}; }, onDidDispose: () => ({}), reveal: () => {} };
        panels.push(panel); return panel;
      },
      onDidChangeTextEditorSelection: () => ({}), showTextDocument: async doc => opened.push(doc.uri.fsPath),
      showErrorMessage: text => messages.push({ type: 'errorDialog', text }), showInformationMessage: text => messages.push({ type: 'infoDialog', text }),
    },
    workspace: { onDidChangeTextDocument: fn => { vscode.changeDocument = fn; return {}; }, fs: { readDirectory: async () => [], stat: async () => ({ type: 1 }) }, openTextDocument: async u => ({ uri: u }) }, env: { openExternal: () => {} },
  };
  return { vscode, commands, panels, opened, messages, context: { subscriptions: [], extensionUri: uri('/extension'), extensionMode: 1 } };
}
const results = {};
const edgeA = core.analyze('a >> p -> x\nb >> q -> y\n');
const edgeB = core.analyze('a >> p -> y\nb >> q -> x\n');
results.edgeDiff = {diagnosticsA:edgeA.diagnostics,diagnosticsB:edgeB.diagnostics,report:core.diffGraphs(edgeA.graph,edgeB.graph,edgeA.frontmatter,edgeB.frontmatter),dot:graphviz.exportDiffDot(edgeA.graph,edgeA.frontmatter,edgeB.graph,edgeB.frontmatter)};
assert.equal(results.edgeDiff.report.addedEdges.length,2);
assert.match(results.edgeDiff.dot, /_nodiff/);
const subflowSource = '---\nbasePath: ../\nprocess:\n  build:\n    subflow: child.pfdsl\n---\na >> build -> b\n';
const docPath = '/repo/.pfdsl/root.pfdsl';
const linkModule = await load('packages/vscode-extension/src/document-link-logic.ts');
const subflow = vscodeHarness(document(docPath, subflowSource));
const preview = await load('packages/vscode-extension/src/preview.ts', { vscode: subflow.vscode });
preview.registerPreview(subflow.context);
await subflow.commands.get('pfdsl.preview')();
subflow.panels[0].receive({ type: 'openFile', path: 'child.pfdsl' });
await new Promise(resolve => setImmediate(resolve));
results.subflow = { diagnostics: core.analyze(subflowSource).diagnostics, editorTarget: linkModule.extractDocumentLinks(subflowSource, docPath)[0].target, previewTarget: subflow.opened[0] };
assert.equal(results.subflow.editorTarget, 'file:///repo/.pfdsl/child.pfdsl');
assert.equal(results.subflow.previewTarget, '/repo/child.pfdsl');
const locations = { quotedArray: '    location: ["docs/a.md", "docs/b.md"]', commaInValue: '    location: ["docs/a,b.md"]', comment: '    location: docs/a.md # explanation', scheme: '    location: ssh://host/path', scalarControl: '    location: "docs/a.md"' };
results.locations = {};
for (const [name, line] of Object.entries(locations)) {
  const source = `---\nartifact:\n  a:\n${line}\n---\na >> build -> b\n`;
