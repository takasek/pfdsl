import * as fs from 'node:fs';
import {join} from 'node:path';
import assert from 'node:assert/strict';
import {deployInstall,readManifest} from '/path/to/pfdsl/.claude/skills/pfd-ops/scripts/check-install-sync.mjs';
import {assemblePluginDistIndependent,assembleCodexAssets} from '/path/to/pfdsl/scripts/lib/gen-plugin.mjs';
const out='/tmp/pfdsl-audit/distribution';
const root=fs.mkdtempSync(join(out,'install-'));
const s=join(root,'skill'),t=join(root,'target');
fs.mkdirSync(join(s,'install'),{recursive:true});fs.mkdirSync(t,{recursive:true});
fs.writeFileSync(join(s,'install/a.txt'),'v1');
const initial=deployInstall(s,t);
const initialManifest=readManifest(t);
fs.writeFileSync(join(s,'install/a.txt'),'v2');
const upgrade=deployInstall(s,t);
const target=fs.readFileSync(join(t,'a.txt'),'utf8');
const upgradeManifest=readManifest(t);
fs.unlinkSync(join(s,'install/a.txt'));
const drop=deployInstall(s,t);
assert.equal(target,'v1');
const lockRoot=fs.mkdtempSync(join(out,'lock-'));
fs.mkdirSync(join(lockRoot,'plugin/pfdsl'),{recursive:true});
fs.mkdirSync(join(lockRoot,'.codex-assets-assembly.lock'));
fs.writeFileSync(join(lockRoot,'AGENTS.md'),'old');
let injected=false,error;
try {assemblePluginDistIndependent({root:lockRoot,pluginRoot:join(lockRoot,'plugin/pfdsl'),deps:{...fs,newRunId:()=> 'contender-b',decodeHarnessCapabilities(){fs.writeFileSync(join(lockRoot,'AGENTS.md'),'published-by-lock-holder-a');injected=true;return [];},assembleClaudeAssets(){return {observed:{'claude-repository':[],'claude-plugin':[]}};},assembleCodexAssets}});} catch(e) {error=e.message;}
const final=fs.readFileSync(join(lockRoot,'AGENTS.md'),'utf8');
assert.equal(injected,true);
assert.equal(final,'old');
const results={install:{initial,initialManifest,upgrade,target,upgradeManifest,drop},lock:{error,injected,final,lockRemains:fs.existsSync(join(lockRoot,'.codex-assets-assembly.lock'))}};
fs.writeFileSync(join(out,'results.json'),JSON.stringify(results,null,2)+'\n');
console.log(JSON.stringify(results,null,2));
