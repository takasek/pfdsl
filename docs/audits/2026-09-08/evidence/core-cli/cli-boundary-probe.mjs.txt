import {mkdtempSync,writeFileSync,readFileSync} from 'node:fs';
import {join} from 'node:path';
import {spawnSync} from 'node:child_process';
import assert from 'node:assert/strict';
import {analyze} from '/path/to/pfdsl/packages/core/dist/index.js';
const root='/path/to/pfdsl';
const out='/tmp/pfdsl-audit/core-cli';
const fixture=mkdtempSync(join(out,'cli-boundary-'));
const file=join(fixture,'format.pfdsl');
const original='"a#b" >> P -> c\n';
writeFileSync(file,original);
const fmt=spawnSync(process.execPath,[join(root,'packages/cli/dist/cli.js'),'fmt',file,'--write'],{cwd:root,encoding:'utf8'});
const after=readFileSync(file,'utf8');
assert.equal(fmt.status,0);assert.equal(analyze(after).edges.length,0);
const a=join(fixture,'a.pfdsl'),b=join(fixture,'b.pfdsl');
writeFileSync(a,'a >> p -> x\nb >> q -> y\n');writeFileSync(b,'a >> p -> y\nb >> q -> x\n');
const run=(format)=>{const p=spawnSync(process.execPath,[join(root,'packages/cli/dist/cli.js'),'diff',a,b,'--format',format],{cwd:root,encoding:'utf8'});return {status:p.status,stdout:p.stdout,stderr:p.stderr};};
const diff={text:run('text'),dot:run('dot')};
assert.equal(diff.dot.status,0);assert.match(diff.dot.stdout,/_nodiff/);
const result={formatWrite:{status:fmt.status,stdout:fmt.stdout,stderr:fmt.stderr,original,after,beforeEdges:analyze(original).edges.length,afterEdges:analyze(after).edges.length},diff};
writeFileSync(join(out,'cli-boundary-results.json'),JSON.stringify(result,null,2)+'\n');
console.log(JSON.stringify(result,null,2));
